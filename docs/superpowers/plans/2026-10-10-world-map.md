```ts
/** The three worlds the game draws a top-down map for, as named in DT_WorldRegionMapData and the BeautyRender folders. */
export const WORLD_KEYS = ["overworld", "sift", "camp"] as const;
export type WorldKey = (typeof WORLD_KEYS)[number];

export interface WorldSource {
  readonly key: WorldKey;
  /** Row name in DT_WorldRegionMapData and folder name under WorldMap/BeautyRender. */
  readonly region: string;
  /** Source images per side: <Region>_<row>_<col>.png. */
  readonly grid: number;
  /** Key in Text/Release/WorldMapLabels.csv for the world's name. */
  readonly label: string;
}

export const WORLDS: readonly WorldSource[] = [
  { key: "overworld", region: "Overworld", grid: 8, label: "worldmap_dimension_overworld" },
  { key: "sift", region: "Sift", grid: 7, label: "worldmap_dimension_wellspring" },
  { key: "camp", region: "Camp", grid: 3, label: "worldmap_dimension_camp" },
];

/** Side of one BeautyRender source image. */
export const SOURCE_TILE = 1024;
/** Side of one published map tile. */
export const TILE = 256;

export const imageSizeOf = (world: Pick<WorldSource, "grid">): number => world.grid * SOURCE_TILE;

/** The zoom at which tiles show the render at full size. */
export const zMaxFor = (imageSize: number): number => Math.max(0, Math.ceil(Math.log2(imageSize / TILE)));
```

`src/lib/map/transform.ts`：

```ts
/**
 * Game position → pixel on a world's stitched BeautyRender, checked against buildings in town (spec §3).
 * The minimap is MapDims blocks square, 1px per block, with the game's Y to the right and X upwards from
 * MinPos at the bottom-left; the render is that square shrunk by FancyMapSizeRatio, still anchored bottom-left.
 */
export interface WorldFrame {
  /** Region bounds in blocks. */
  readonly minX: number;
  readonly maxX: number;
  readonly minY: number;
  readonly maxY: number;
  /** MapDims: side of the minimap square in blocks. */
  readonly span: number;
  /** FancyMapSizeRatio. */
  readonly ratio: number;
  /** Side of the stitched render in pixels. */
  readonly imageSize: number;
}

export interface BlockPoint {
  readonly x: number;
  readonly y: number;
}

export interface ImagePoint {
  readonly px: number;
  readonly py: number;
}

const CM_PER_BLOCK = 100;

export const toBlocks = (location: { readonly X: number; readonly Y: number }): BlockPoint => ({
  x: location.X / CM_PER_BLOCK,
  y: location.Y / CM_PER_BLOCK,
});

export const isInside = (frame: WorldFrame, block: BlockPoint): boolean =>
  block.x >= frame.minX && block.x <= frame.maxX && block.y >= frame.minY && block.y <= frame.maxY;

export function toImagePixel(frame: WorldFrame, block: BlockPoint): ImagePoint {
  const u = ((block.y - frame.minY) / frame.span) * frame.ratio;
  const v = 1 - ((block.x - frame.minX) / frame.span) * frame.ratio;
  return { px: u * frame.imageSize, py: v * frame.imageSize };
}
```

`src/lib/map/markers.ts`：

```ts
import { MARKER_KINDS, type MapMarker, type MarkerKind } from "../../../scripts/extract/map-schema";

const DEFAULT_LIMIT = 20;

const normalize = (text: string): string => text.trim().toLocaleLowerCase();

const searchableText = (marker: MapMarker): string[] =>
  [marker.name.zh, marker.name.en, marker.area?.zh, marker.area?.en]
    .filter((text): text is string => text !== undefined)
    .map(normalize);

/** Markers whose name or area contains the query, in their original order. */
export function searchMarkers(markers: readonly MapMarker[], query: string, limit = DEFAULT_LIMIT): MapMarker[] {
  const needle = normalize(query);
  if (needle === "") return [];
  return markers.filter((marker) => searchableText(marker).some((text) => text.includes(needle))).slice(0, limit);
}

export const countByKind = (markers: readonly MapMarker[]): Record<MarkerKind, number> =>
  Object.fromEntries(MARKER_KINDS.map((kind) => [kind, markers.filter((marker) => marker.kind === kind).length])) as Record<
    MarkerKind,
    number
  >;

const isKind = (value: unknown): value is MarkerKind =>
  typeof value === "string" && (MARKER_KINDS as readonly string[]).includes(value);

/** Hidden kinds saved in storage; anything unreadable means nothing is hidden. */
export function parseHiddenKinds(raw: string | null): Set<MarkerKind> {
  if (!raw) return new Set();
  try {
    const value: unknown = JSON.parse(raw);
    return Array.isArray(value) ? new Set(value.filter(isKind)) : new Set();
  } catch {
    return new Set();
  }
}

export const serializeHiddenKinds = (hidden: ReadonlySet<MarkerKind>): string =>
  JSON.stringify(MARKER_KINDS.filter((kind) => hidden.has(kind)));

export function withKind(hidden: ReadonlySet<MarkerKind>, kind: MarkerKind, visible: boolean): Set<MarkerKind> {
  return new Set([...hidden].filter((other) => other !== kind).concat(visible ? [] : [kind]));
}
```

```ts
import type { WorldKey } from "./worlds";

/** A map view: zoom level and centre, as pixels on the full-size render. */
export interface View {
  readonly z: number;
  readonly px: number;
  readonly py: number;
}

export interface ViewLimits {
  readonly zMin: number;
  readonly zMax: number;
  readonly imageSize: number;
}

const NUMBER = "(-?\\d+(?:\\.\\d+)?)";
const VIEW_HASH = new RegExp(`^#${NUMBER}/${NUMBER}/${NUMBER}$`);

const clamp = (value: number, low: number, high: number): number => Math.min(high, Math.max(low, value));

export function parseWorld(search: string, available: readonly WorldKey[]): WorldKey | undefined {
  const wanted = new URLSearchParams(search).get("world");
  return available.find((key) => key === wanted) ?? available[0];
}

/** The view in a `#z/px/py` hash, pulled back inside the map; undefined when the hash is not a view. */
export function parseView(hash: string, limits: ViewLimits): View | undefined {
  const match = VIEW_HASH.exec(hash);
  if (!match) return undefined;
  const [z, px, py] = match.slice(1).map(Number);
  return {
    z: clamp(Math.round(z), limits.zMin, limits.zMax),
    px: clamp(px, 0, limits.imageSize),
    py: clamp(py, 0, limits.imageSize),
  };
}

export const formatView = (view: View): string => `#${Math.round(view.z)}/${Math.round(view.px)}/${Math.round(view.py)}`;

export function worldSearch(search: string, world: WorldKey): string {
  const params = new URLSearchParams(search);
  params.set("world", world);
  return `?${params.toString()}`;
}
```

```ts
import type { Lang } from "./config";
import type { LocaleFile, Localized } from "./schemas";

const LOCTABLE = /^LOCTABLE\("([^"]+)",\s*"([^"]+)"\)$/;

export interface LocRef {
  readonly namespace: string;
  readonly key: string;
}

/** `LOCTABLE("VendorLabels.csv", "key")` → the locale file namespace and key it points at. */
export function locRef(text: string): LocRef | undefined {
  const match = LOCTABLE.exec(text.trim());
  return match ? { namespace: `Text/Release/${match[1]}`, key: match[2] } : undefined;
}

/** Text in both languages; Chinese falls back to English, and no English means no text. */
export function localize(locales: Readonly<Record<Lang, LocaleFile>>, namespace: string, key: string): Localized | undefined {
  const en = locales.en[namespace]?.[key];
  if (!en) return undefined;
  return { en, zh: locales.zh[namespace]?.[key] || en };
}
```

```ts
/** The DataTables the world map reads, validated so a game update that renames a field fails loudly. */
import { existsSync } from "node:fs";
import path from "node:path";
import { z } from "zod";
import { DATATABLES_DIR } from "./game-tables";
import { ExtractError, formatZodIssue } from "./issues";
import { readJsonFile } from "./read-json";

const Tag = z.object({ TagName: z.string() });
const Vec = z.object({ X: z.number(), Y: z.number(), Z: z.number() });

const RegionRow = z.object({
  Name: z.string(),
  RegionData: z.object({
    FancyMapSizeRatio: z.object({ X: z.number() }),
    Bounds: z.object({ MapDims: z.object({ X: z.number() }), MinPos: Vec, MaxPos: Vec }),
  }),
});
const TeleportRow = z.object({ Name: z.string(), AreaTag: Tag, IdTag: Tag, Type: z.string(), Location: Vec });
const ExplorationRow = z.object({ Name: z.string(), TeleportTag: Tag });
const MerchantRow = z.object({ Name: z.string(), Location: Vec, MerchantName: z.string(), MerchantDescription: z.string() });
const AreaRow = z.object({
  Name: z.string(),
  TypeTag: Tag,
  TypeTextData: z.object({ Texts: z.record(z.string(), z.object({ Text: z.string() })) }).optional(),
  AreaIcon: z.string().optional(),
});

export const MapTablesSchema = z.object({
  regions: z.array(RegionRow),
  teleports: z.array(TeleportRow),
  exploration: z.array(ExplorationRow),
  merchants: z.array(MerchantRow),
  areas: z.array(AreaRow),
});
export type MapTables = z.infer<typeof MapTablesSchema>;
export type TeleportRowValue = z.infer<typeof TeleportRow>;
export type AreaRowValue = z.infer<typeof AreaRow>;

const FILES: Readonly<Record<keyof MapTables, string>> = {
  regions: "DT_WorldRegionMapData",
  teleports: "DT_WorldTeleportPoints",
  exploration: "DT_WorldExplorationTeleportPoints",
  merchants: "DT_MerchantLocationsDefinition",
  areas: "DT_AreaDefinition",
};

/** Reads the map tables; undefined when source-data has no region table (the map is then left empty). */
export async function loadMapTables(sourceRoot: string): Promise<MapTables | undefined> {
  const dir = path.join(sourceRoot, ...DATATABLES_DIR.split("/"));
  if (!existsSync(path.join(dir, `${FILES.regions}.json`))) return undefined;
  const entries = await Promise.all(
    Object.entries(FILES).map(async ([key, name]) => {
      const rows = await readJsonFile(path.join(dir, `${name}.json`), `資料表 ${name}`);
      if (!Array.isArray(rows)) throw new ExtractError(`資料表 ${name} 不存在或不是陣列：${dir}`);
      return [key, rows] as const;
    }),
  );
  const parsed = MapTablesSchema.safeParse(Object.fromEntries(entries));
  if (!parsed.success) {
    throw new ExtractError(`地圖資料表欄位與預期不符（遊戲更新了格式？）：${parsed.error.issues.slice(0, 3).map(formatZodIssue).join("；")}`);
  }
  return parsed.data;
}
```

```ts
function makeContext(tables: MapTables, locales: Locales): Context {
  const areas = new Map<string, AreaRowValue>(tables.areas.map((row) => [lower(row.TypeTag.TagName), row]));
  const text = (namespace: string, key: string) => localize(locales, namespace, key);
  const loc = (value: string) => {
    const ref = locRef(value);
    return ref ? text(ref.namespace, ref.key) : undefined;
  };
  const areaName = (tag: string) => {
    const texts = Object.entries(areas.get(lower(tag))?.TypeTextData?.Texts ?? {});
    const name = texts.find(([key]) => NAME_TEXT.test(key))?.[1].Text;
    return name ? loc(name) : undefined;
  };
  const areaIcon = (tag: string) => {
    const match = /\/AreaIcons\/([^./]+)\./.exec(areas.get(lower(tag))?.AreaIcon ?? "");
    return match ? `${AREA_ICON_DIR}/${match[1]}.png` : undefined;
  };

function worldFrames(tables: MapTables, ctx: Context): { frames: Frame[]; issues: Issue[] } {
  const found = WORLDS.map((world) => ({ world, row: tables.regions.find((row) => row.Name === world.region) }));
  return {
    frames: found.flatMap(({ world, row }) => {
      if (!row) return [];

      const frame: WorldFrame = {
        minX: MinPos.X,
        maxX: MaxPos.X,

function toMarker(candidate: Candidate, frame: WorldFrame, iconExists: (relative: string) => boolean): MapMarker {
  const { px, py } = toImagePixel(frame, candidate.block);
  const icon = candidate.iconSource && iconExists(candidate.iconSource) ? publishedIcon(candidate.iconSource) : undefined;
  return {
    id: candidate.id,
    kind: candidate.kind,
    px: round1(px),
    py: round1(py),
    name: candidate.name,
    ...optional("area", candidate.area),
    ...optional("note", candidate.note),
    ...optional("icon", icon),
  };
}

function areaLabels(points: readonly { candidate: Candidate; marker: MapMarker }[], ctx: Context, iconExists: (relative: string) => boolean): MapMarker[] {
  const tags = [...new Set(points.flatMap(({ candidate }) => (candidate.areaTag ? [lower(candidate.areaTag)] : [])))];
  return tags.flatMap((tag) => {
    const name = ctx.areaName(tag);
    if (!name) return [];
    const members = points.filter(({ candidate }) => candidate.areaTag && lower(candidate.areaTag) === tag).map(({ marker }) => marker);
    const mean = (pick: (marker: MapMarker) => number) => round1(members.reduce((sum, marker) => sum + pick(marker), 0) / members.length);
    const source = ctx.areaIcon(tag);
    const icon = source && iconExists(source) ? publishedIcon(source) : undefined;
    return [{ id: `area:${tag}`, kind: "area" as const, px: mean((m) => m.px), py: mean((m) => m.py), name, ...optional("icon", icon) }];
  });
}

const byKindThenId = (a: MapMarker, b: MapMarker): number => kindOrder(a.kind) - kindOrder(b.kind) || a.id.localeCompare(b.id, "en");

function buildWorld(frame: Frame, candidates: readonly Candidate[], ctx: Context, iconExists: (relative: string) => boolean): WorldMap {
  const points = candidates.map((candidate) => ({ candidate, marker: toMarker(candidate, frame.frame, iconExists) }));
  return {
    key: frame.world.key,
    name: frame.name,
    frame: frame.frame,
    zMax: zMaxFor(frame.frame.imageSize),
    markers: [...points.map(({ marker }) => marker), ...areaLabels(points, ctx, iconExists)].sort(byKindThenId),
  };
}

function iconsOf(candidates: readonly Candidate[], ctx: Context, iconExists: (relative: string) => boolean): MapIcon[] {
  const sources = candidates.flatMap((candidate) => [candidate.iconSource, candidate.areaTag ? ctx.areaIcon(candidate.areaTag) : undefined]);
  const present = [...new Set(sources.filter((source): source is string => source !== undefined && iconExists(source)))];
  return present.map((from) => ({ from, published: publishedIcon(from) }));
}

export function buildMapData(tables: MapTables, locales: Locales, iconExists: (relative: string) => boolean): MapResult {
  const ctx = makeContext(tables, locales);
  const { frames, issues: frameIssues } = worldFrames(tables, ctx);
  const gathered = merge([minecarts(tables, ctx), rifts(tables, ctx), portals(tables, ctx), merchants(tables, ctx)]);
  const inside = (candidate: Candidate) => frames.some((frame) => isInside(frame.frame, candidate.block));
  const placedCandidates = gathered.candidates.filter(inside);
  return {
    data: {
      worlds: frames.map((frame) => buildWorld(frame, placedCandidates.filter((c) => isInside(frame.frame, c.block)), ctx, iconExists)),
    },
    icons: iconsOf(placedCandidates, ctx, iconExists),
    issues: [
      ...frameIssues,
      ...gathered.issues,
      ...gathered.candidates.filter((c) => !inside(c)).map((c) => issue("map-marker-outside", `${c.kind}:${c.id}`)),
    ],
  };
}
```

```ts
/** Tile arithmetic for the map pyramid: zMax holds the render at full size, each level below halves it. */
import { SOURCE_TILE, TILE } from "../../src/lib/map/worlds";

export interface TileId {
  readonly z: number;
  readonly x: number;
  readonly y: number;
}

const CUTS_PER_SOURCE = SOURCE_TILE / TILE;

export const levelSize = (imageSize: number, zMax: number, z: number): number => Math.ceil(imageSize / 2 ** (zMax - z));

export const tilesPerSide = (imageSize: number, zMax: number, z: number): number => Math.ceil(levelSize(imageSize, zMax, z) / TILE);

export function levelTiles(imageSize: number, zMax: number, z: number): TileId[] {
  const side = tilesPerSide(imageSize, zMax, z);
  return Array.from({ length: side * side }, (_, index) => ({ z, x: index % side, y: Math.floor(index / side) }));
}

export const childrenOf = (tile: TileId): { tile: TileId; left: number; top: number }[] =>
  [0, 1].flatMap((dy) =>
    [0, 1].map((dx) => ({ tile: { z: tile.z + 1, x: tile.x * 2 + dx, y: tile.y * 2 + dy }, left: dx * TILE, top: dy * TILE })),
  );

/** Which source image a top-level tile comes from, and where in it. */
export const sourceCut = (tile: TileId): { row: number; col: number; left: number; top: number } => ({
  row: Math.floor(tile.y / CUTS_PER_SOURCE),
  col: Math.floor(tile.x / CUTS_PER_SOURCE),
  left: (tile.x % CUTS_PER_SOURCE) * TILE,
  top: (tile.y % CUTS_PER_SOURCE) * TILE,
});

const pad2 = (value: number): string => String(value).padStart(2, "0");

export const sourceName = (region: string, row: number, col: number): string => `${region}_${pad2(row)}_${pad2(col)}.png`;
```

```ts
/** Cuts a world's BeautyRender images into the published webp tile pyramid. */
import { existsSync } from "node:fs";
import { mkdir, readFile, rm, stat, writeFile } from "node:fs/promises";
import path from "node:path";
import sharp from "sharp";
import { TILE, imageSizeOf, zMaxFor } from "../../src/lib/map/worlds";
import { childrenOf, levelTiles, sourceCut, sourceName, tileKey, type TileId } from "./pyramid";

export interface TileWorld {
  readonly key: string;
  readonly region: string;
  readonly grid: number;
}

export interface TileOptions {
  /** WorldMap/BeautyRender: one folder per region. */
  readonly sourceDir: string;
  /** public/map. */
  readonly outDir: string;
  readonly quality: number;
}

export interface TileSummary {
  readonly world: string;
  readonly tiles: number;
  readonly bytes: number;
  readonly skipped: boolean;
}

export class MapTilesError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "MapTilesError";
  }
}

interface SourceStamp {
  readonly name: string;
  readonly size: number;
  readonly mtimeMs: number;
}

type Level = ReadonlyMap<string, Buffer>;

const TRANSPARENT = { r: 0, g: 0, b: 0, alpha: 0 };

async function stampSources(world: TileWorld, sourceDir: string): Promise<SourceStamp[]> {
  const names = Array.from({ length: world.grid * world.grid }, (_, i) => sourceName(world.region, Math.floor(i / world.grid), i % world.grid));
  const missing = names.filter((name) => !existsSync(path.join(sourceDir, world.region, name)));
  if (missing.length > 0) throw new MapTilesError(`${world.region} 缺少地圖原圖：${missing.join("、")}`);
  return Promise.all(
    names.map(async (name) => {
      const info = await stat(path.join(sourceDir, world.region, name));
      return { name, size: info.size, mtimeMs: Math.round(info.mtimeMs) };
    }),
  );
}

async function sameSources(manifestPath: string, stamps: readonly SourceStamp[]): Promise<boolean> {
  try {
    const manifest = JSON.parse(await readFile(manifestPath, "utf8")) as { sources?: unknown };
    return JSON.stringify(manifest.sources) === JSON.stringify(stamps);
  } catch {
    return false;
  }
}

async function cutSource(file: string, tiles: readonly TileId[], quality: number): Promise<[string, Buffer][]> {
  const { data, info } = await sharp(file).ensureAlpha().raw().toBuffer({ resolveWithObject: true });
  return Promise.all(
    tiles.map(async (tile): Promise<[string, Buffer]> => {
      const { left, top } = sourceCut(tile);
      const webp = await sharp(data, { raw: info }).extract({ left, top, width: TILE, height: TILE }).webp({ quality }).toBuffer();
      return [tileKey(tile), webp];
    }),
  );
}

async function topLevel(world: TileWorld, options: TileOptions, imageSize: number, zMax: number): Promise<Level> {
  const tiles = levelTiles(imageSize, zMax, zMax);
  const names = [...new Set(tiles.map((tile) => { const cut = sourceCut(tile); return sourceName(world.region, cut.row, cut.col); }))];
  const parts: [string, Buffer][][] = [];
  for (const name of names) {
    const own = tiles.filter((tile) => { const cut = sourceCut(tile); return sourceName(world.region, cut.row, cut.col) === name; });
    parts.push(await cutSource(path.join(options.sourceDir, world.region, name), own, options.quality));
  }
  return new Map(parts.flat());
}

async function downscale(children: Level, tile: TileId, quality: number): Promise<Buffer> {
  const parts = childrenOf(tile).flatMap(({ tile: child, left, top }) => {
    const input = children.get(tileKey(child));
    return input ? [{ input, left, top }] : [];
  });
  const square = await sharp({ create: { width: TILE * 2, height: TILE * 2, channels: 4, background: TRANSPARENT } })
    .composite(parts)
    .png()
    .toBuffer();
  return sharp(square).resize(TILE, TILE).webp({ quality }).toBuffer();
}

async function buildLevels(world: TileWorld, options: TileOptions, imageSize: number, zMax: number): Promise<Level[]> {
  const levels: Level[] = [await topLevel(world, options, imageSize, zMax)];
  for (let z = zMax - 1; z >= 0; z -= 1) {
    const children = levels[levels.length - 1];
    const tiles = levelTiles(imageSize, zMax, z);
    levels.push(new Map(await Promise.all(tiles.map(async (tile): Promise<[string, Buffer]> => [tileKey(tile), await downscale(children, tile, options.quality)]))));
  }
  return levels;
}

async function writeLevels(worldDir: string, levels: readonly Level[]): Promise<{ count: number; bytes: number }> {
  const tiles = levels.flatMap((level) => [...level]);
  for (const [key, buffer] of tiles) {
    const file = path.join(worldDir, ...key.split("/")) + ".webp";
    await mkdir(path.dirname(file), { recursive: true });
    await writeFile(file, buffer);
  }
  return { count: tiles.length, bytes: tiles.reduce((sum, [, buffer]) => sum + buffer.length, 0) };
}

export async function buildWorldTiles(world: TileWorld, options: TileOptions): Promise<TileSummary> {
  const stamps = await stampSources(world, options.sourceDir);
  const worldDir = path.join(options.outDir, world.key);
  const manifestPath = path.join(worldDir, "manifest.json");
  if (await sameSources(manifestPath, stamps)) return { world: world.key, tiles: 0, bytes: 0, skipped: true };
  const imageSize = imageSizeOf(world);
  const zMax = zMaxFor(imageSize);
  const levels = await buildLevels(world, options, imageSize, zMax);
  await rm(worldDir, { recursive: true, force: true });
  const written = await writeLevels(worldDir, levels);
  await writeFile(manifestPath, `${JSON.stringify({ imageSize, zMax, sources: stamps }, null, 2)}\n`, "utf8");
  return { world: world.key, tiles: written.count, bytes: written.bytes, skipped: false };
}
```

```ts
/** DOM for map pins, popups and search results, built with textContent so game text is never parsed as HTML. */
import type { MapMarker } from "../../../scripts/extract/map-schema";
import type { Lang } from "../i18n";
import { assetUrl } from "../url";

function element<K extends keyof HTMLElementTagNameMap>(tag: K, className: string, text?: string): HTMLElementTagNameMap[K] {
  const node = document.createElement(tag);
  node.className = className;
  if (text !== undefined) node.textContent = text;
  return node;
}

function iconImage(path: string, className: string): HTMLImageElement {
  const image = element("img", className);
  image.src = assetUrl(path);
  image.alt = "";
  return image;
}

export function pinElement(marker: MapMarker, lang: Lang): HTMLElement {
  if (marker.kind !== "area") {
    const pin = element("span", "map-pin");
    if (marker.icon) pin.append(iconImage(marker.icon, "map-pin-icon"));
    return pin;
  }
  const label = element("span", "map-label");
  if (marker.icon) label.append(iconImage(marker.icon, "map-label-icon"));
  label.append(element("span", "map-label-text", marker.name[lang]));
  return label;
}

export function popupElement(marker: MapMarker, lang: Lang, kindLabel: string): HTMLElement {
  const popup = element("div", "map-popup");
  popup.append(element("p", "map-popup-kind mono", kindLabel), element("p", "map-popup-name", marker.name[lang]));
  if (marker.area) popup.append(element("p", "map-popup-area", marker.area[lang]));
  if (marker.note) popup.append(element("p", "map-popup-note", marker.note[lang]));
  return popup;
}

export function resultElement(marker: MapMarker, lang: Lang, kindLabel: string): HTMLLIElement {
  const item = element("li", "");
  const button = element("button", "map-result", [marker.name[lang], kindLabel, marker.area?.[lang]].filter(Boolean).join(" · "));
  button.type = "button";
  button.dataset.markerId = marker.id;
  item.append(button);
  return item;
}
```

```ts
/** The map page's client: Leaflet in CRS.Simple over the tile pyramid, plus the filter and search panel. */
import L from "leaflet";
import "leaflet/dist/leaflet.css";
import { MARKER_KINDS, type MapMarker, type MarkerKind, type WorldMap } from "../../../scripts/extract/map-schema";
import type { Lang } from "../i18n";
import { pinElement, popupElement, resultElement } from "./marker-dom";
import { countByKind, parseHiddenKinds, searchMarkers, serializeHiddenKinds, withKind } from "./markers";
import { tileUrlTemplate } from "./tiles";
import { formatView, parseView, parseWorld, worldSearch } from "./view-state";
import { TILE } from "./worlds";

const STORAGE_KEY = "mcd2-map-hidden";
/** Area names show from this many levels below the top zoom. */
const LABEL_LEVELS = 3;
/** Search results open no further out than one level below the top zoom. */
const FOCUS_LEVELS = 1;
const PIN_SIZE: L.PointTuple = [16, 16];

interface Payload {
  readonly worlds: readonly WorldMap[];
  readonly labels: Readonly<Record<MarkerKind, string>>;
}

interface Shown {
  readonly world: WorldMap;
  readonly map: L.Map;
  readonly layers: ReadonlyMap<MarkerKind, L.LayerGroup>;
  readonly pins: ReadonlyMap<string, L.Marker>;
}

/** A pixel on the full-size render as a LatLng: CRS.Simple maps one unit to 2^z pixels. */
const toLatLng = (world: WorldMap, px: number, py: number): L.LatLng => L.latLng(-py / 2 ** world.zMax, px / 2 ** world.zMax);

function readHidden(): Set<MarkerKind> {
  try {
    return parseHiddenKinds(localStorage.getItem(STORAGE_KEY));
  } catch {
    return new Set();
  }
}

function writeHidden(hidden: ReadonlySet<MarkerKind>): void {
  try {
    localStorage.setItem(STORAGE_KEY, serializeHiddenKinds(hidden));
  } catch {
    // Storage unavailable: the choice lasts for this page view only.
  }
}

function replaceLocation(search: string, hash: string): void {
  history.replaceState(history.state, "", `${location.pathname}${search}${hash}`);
  window.dispatchEvent(new Event("mcd2:locationchange"));
}

function addPins(map: L.Map, world: WorldMap, lang: Lang, labels: Payload["labels"], hidden: ReadonlySet<MarkerKind>) {
  const layers = new Map(MARKER_KINDS.map((kind) => [kind, L.layerGroup()] as const));
  const pins = new Map<string, L.Marker>();
  for (const marker of world.markers) {
    const icon = L.divIcon({
      className: `map-marker map-marker--${marker.kind}`,
      html: pinElement(marker, lang),
      iconSize: marker.kind === "area" ? [0, 0] : PIN_SIZE,
    });
    const pin = L.marker(toLatLng(world, marker.px, marker.py), { icon, title: marker.name[lang], keyboard: marker.kind !== "area" });
    pin.bindPopup(popupElement(marker, lang, labels[marker.kind]));
    layers.get(marker.kind)?.addLayer(pin);
    pins.set(marker.id, pin);
  }
  for (const [kind, layer] of layers) if (!hidden.has(kind)) layer.addTo(map);
  return { layers, pins };
}

function trackView(root: HTMLElement, map: L.Map, world: WorldMap): void {
  const update = () => {
    const zoom = map.getZoom();
    const centre = map.project(map.getCenter(), world.zMax);
    root.dataset.zoom = String(zoom);
    root.dataset.labels = zoom >= world.zMax - LABEL_LEVELS ? "on" : "off";
    replaceLocation(location.search, formatView({ z: zoom, px: centre.x, py: centre.y }));
  };
  map.on("moveend", update);
  update();
}

function showWorld(root: HTMLElement, canvas: HTMLElement, world: WorldMap, payload: Payload, hidden: ReadonlySet<MarkerKind>, useHash: boolean): Shown {
  const lang = root.dataset.lang as Lang;
  const size = world.frame.imageSize;
  const bounds = L.latLngBounds(toLatLng(world, 0, size), toLatLng(world, size, 0));
  const map = L.map(canvas, { crs: L.CRS.Simple, minZoom: 0, maxZoom: world.zMax, zoomSnap: 1, maxBounds: bounds.pad(0.25), attributionControl: false });
  L.tileLayer(tileUrlTemplate(import.meta.env.BASE_URL, world.key), { tileSize: TILE, maxNativeZoom: world.zMax, maxZoom: world.zMax, noWrap: true, bounds }).addTo(map);
  const { layers, pins } = addPins(map, world, lang, payload.labels, hidden);
  const view = useHash ? parseView(location.hash, { zMin: 0, zMax: world.zMax, imageSize: size }) : undefined;
  if (view) map.setView(toLatLng(world, view.px, view.py), view.z, { animate: false });
  else if (world.markers.length > 0) map.fitBounds(L.latLngBounds(world.markers.map((m) => toLatLng(world, m.px, m.py))).pad(0.1), { animate: false });
  else map.fitBounds(bounds, { animate: false });
  trackView(root, map, world);
  root.dataset.world = world.key;
  return { world, map, layers, pins };
}

function syncPanel(root: HTMLElement, world: WorldMap, hidden: ReadonlySet<MarkerKind>): void {
  const counts = countByKind(world.markers);
  for (const link of root.querySelectorAll<HTMLAnchorElement>(".map-tabs a[data-world]")) {
    if (link.dataset.world === world.key) link.setAttribute("aria-current", "page");
    else link.removeAttribute("aria-current");
  }
  for (const kind of MARKER_KINDS) {
    const count = root.querySelector(`[data-count="${kind}"]`);
    if (count) count.textContent = String(counts[kind]);
    const box = root.querySelector<HTMLInputElement>(`input[name="kind"][value="${kind}"]`);
    if (box) box.checked = !hidden.has(kind);
  }
}

function renderResults(root: HTMLElement, shown: Shown, labels: Payload["labels"], query: string, focus: (marker: MapMarker) => void): void {
  const lang = root.dataset.lang as Lang;
  const list = root.querySelector<HTMLUListElement>("[data-testid='map-results']");
  const empty = root.querySelector<HTMLElement>("[data-testid='map-empty']");
  if (!list || !empty) return;
  const found = searchMarkers(shown.world.markers, query);
  list.replaceChildren(...found.map((marker) => resultElement(marker, lang, labels[marker.kind])));
  empty.hidden = query.trim() === "" || found.length > 0;
  for (const button of list.querySelectorAll<HTMLButtonElement>("button[data-marker-id]")) {
    const marker = found.find((m) => m.id === button.dataset.markerId);
    if (marker) button.addEventListener("click", () => focus(marker));
  }
}

export function mountWorldMap(root: HTMLElement): void {
  const payload = JSON.parse(root.querySelector("[data-map-json]")?.textContent ?? "{}") as Payload;
  const canvas = root.querySelector<HTMLElement>("[data-map-canvas]");
  const keys = (payload.worlds ?? []).map((world) => world.key);
  const initial = payload.worlds?.find((world) => world.key === parseWorld(location.search, keys));
  const search = root.querySelector<HTMLInputElement>("[data-testid='map-search']");
  if (!canvas || !initial) return;
  let hidden = readHidden();
  let shown: Shown;
  try {
    shown = showWorld(root, canvas, initial, payload, hidden, true);
  } catch (error) {
    console.error("World map failed to start", error);
    return;
  }
  syncPanel(root, shown.world, hidden);
  root.dataset.ready = "true";

  const setKind = (kind: MarkerKind, visible: boolean) => {
    hidden = withKind(hidden, kind, visible);
    writeHidden(hidden);
    const layer = shown.layers.get(kind);
    if (layer && visible) layer.addTo(shown.map);
    if (layer && !visible) layer.remove();
    syncPanel(root, shown.world, hidden);
  };

  const focus = (marker: MapMarker) => {
    if (hidden.has(marker.kind)) setKind(marker.kind, true);
    const zoom = Math.max(shown.map.getZoom(), shown.world.zMax - FOCUS_LEVELS);
    shown.map.setView(toLatLng(shown.world, marker.px, marker.py), zoom);
    shown.pins.get(marker.id)?.openPopup();
  };

  for (const box of root.querySelectorAll<HTMLInputElement>("input[name='kind']")) {
    box.addEventListener("change", () => setKind(box.value as MarkerKind, box.checked));
  }

  for (const link of root.querySelectorAll<HTMLAnchorElement>(".map-tabs a[data-world]")) {
    link.addEventListener("click", (event) => {
      const next = payload.worlds.find((world) => world.key === link.dataset.world);
      if (!next) return;
      event.preventDefault();
      if (next.key === shown.world.key) return;
      shown.map.remove();
      replaceLocation(worldSearch(location.search, next.key), "");
      shown = showWorld(root, canvas, next, payload, hidden, false);
      syncPanel(root, shown.world, hidden);
      if (search) search.value = "";
      renderResults(root, shown, payload.labels, "", focus);
    });
  }

  search?.addEventListener("input", () => renderResults(root, shown, payload.labels, search.value, focus));
}
```
