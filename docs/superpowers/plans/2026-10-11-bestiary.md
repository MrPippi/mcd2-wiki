# 怪物圖鑑與取得方式 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增 16 語言的怪物圖鑑（總覽頁加每隻怪物一頁，含遊戲模型渲染的頭像）與物品頁的「取得方式」，全部資料由 `npm run extract` 從 `source-data` 的 DataTables 抽出。

**Architecture:** 抽取端先把 DataTables 讀進 zod 驗證過的型別（Task 1），再依序算出怪物主檔（Task 2）、出沒區域（Task 3）、頭像（Task 4）、掉落規則（Task 5）、物品池與物品索引（Task 6）、物品反查來源（Task 7），最後由 `bestiary.ts` 串進 `pipeline.ts` 並寫出 `mobs.json`、`loot-sources.json`（Task 8）。網站端只讀這兩個 JSON：先補齊 16 語言字串（Task 9），再做怪物頁（Task 10）、物品頁「取得方式」（Task 11），最後接選單、搜尋、E2E 與全面驗證（Task 12）。

**Tech Stack:** Astro 7、TypeScript、zod 4、Vitest 5、Playwright 1.64、Pagefind 1.5、sharp 0.35。

**Spec:** `docs/superpowers/specs/2026-10-11-bestiary-design.md`（已含頭像修訂，commit `26b274f`）。

## Review Focus

1. **名稱鍵必須從 `TypeTextData` 解析，不可由列名推導**：`SlimeLarge`／`SlimeMedium`／`SlimeSmall` 都叫「Slime」（同一個鍵 `SW_Mob_Slime_…`）→ 合併成一隻、一頁；`ZombieFrozen` 的鍵開頭是 `Sw_`（小寫 w）、TypeTag 是 `Sw.Mob.ZombieFrozen` → 仍要找得到名字並與 `SW.Mob.ZombieFrozen` 視為同一標籤。（Task 2）
2. **靈魂腐化版本併入原怪物**，不另開頁、不出現在總覽；原怪物頁的「靈魂腐化」小節只在遊戲真的有這個版本時出現，其掉落是 0 必掉 + 1 次 @25%。（Task 2、5）
3. **套裝標籤展開**：區域池／任務裡的 `SW.Item.ArmorSet.Mushroom` 要展開成四個部位，每個部位的物品頁都有「取得方式」；`_Unique` 套裝標籤不可混進一般件；任務「4 選 2 必得」要標出 2/4。（Task 6、7）
4. **`Operation: Multiply` 且權重 0 = 排除**，不是「加入」：`CampA1` 只有一個 Multiply 0 → 池是空的；`AllEnchantmentBooks` 排除 Piercing／Radiance／SoulInfusedPotion，但任務 `EnchantsmithBookReward` 仍然列出它們。商人池的 `SW.Item.Property.Unique` Multiply 0 → 傳奇（Unique）物品不會出現在商人。（Task 6、7）
5. **商人只顯示類別**（「商人（近戰武器類）」），不展開成每件物品的來源清單以外的內容；商人不能出現在沒有該類別 trait 的物品上。（Task 7）
6. **遙測金鑰永遠不在輸出**：抽取讀取清單不含 `OnlineConstants`，`src/data/*.json` 找不到其中任何長度 ≥ 20 的字串值。（Task 1、8）
7. **對不上的東西要進報告，不能靜默丟掉也不能顯示錯的內容**：未對應的波次前綴、找不到名稱的區域、找不到掉落列的怪物、拼法不一致的標籤（`Springstone`／`SpringStone`）、有錯字的任務列名（`QuestMeleeDaggerUnique_)1`）、沒有對上怪物的渲染圖（`Evoker_FangTrap` 等）。頁面只是少一段。（Task 3、4、5、7、8）
8. **頭像只從 `CharacterReference` 對**（不用列名，列名只對得上 32/86）；靈魂腐化沿用原怪物頭像；灰色無貼圖的 `Geomancer_BombPillar.png` 永不發布；手動覆蓋檔優先於渲染圖；沒有圖的怪物顯示預留框。（Task 4、10、12）
9. **機率不以百分比出現在物品頁**；怪物頁只描述「次數與每次機率」，且每頁都有「依遊戲資料整理，實際掉落可能受遊戲機制影響」。德文、俄文等長語言在手機寬度下不得橫向溢出。（Task 10、11、12）

/** Small hand-made mob and loot DataTable rows in the game's export shape. Every builder output is validated by the real schema. */
import { LootTablesSchema, type LootTables } from "../../scripts/extract/loot-tables";
import { MobTablesSchema, type MobTables } from "../../scripts/extract/mob-tables";

const tag = (TagName: string) => ({ TagName });
export const tags = (...names: string[]) => ({ GameplayTags: names.map(tag) });
const nameText = (table: string, key: string) => ({
  Texts: { '(TagName="SW.TextEntry.Name")': { Text: `LOCTABLE("${table}.csv", "${key}")` } },
});

export interface MobOptions {
  readonly key?: string;            // name key; default SW_Mob_<Name>_SW_TextEntry_Name
  readonly traits?: readonly string[];
  readonly attrs?: Readonly<Record<string, number>>;  // "Health" -> SW.Attribute.Health
  readonly reference?: string;      // CharacterReference
  readonly variants?: readonly string[];              // Variants.Replacements tags
  readonly unnamed?: boolean;
}

export function mobRow(table: string, name: string, typeTag: string, options: MobOptions = {}) {
  return {
    Name: name,
    TypeTag: tag(typeTag),
    Traits: tags(...(options.traits ?? [])),
    CharacterReference: options.reference ?? `/Game/Art/Characters/${name}/BP_${name}.BP_${name}_C`,
    AttributeValues: Object.entries(options.attrs ?? {}).map(([attribute, Value]) => ({ Attribute: tag(`SW.Attribute.${attribute}`), Value })),
    ...(options.unnamed ? {} : { TypeTextData: nameText(table, options.key ?? `SW_Mob_${name}_SW_TextEntry_Name`) }),
    ...(options.variants ? { Variants: { Replacements: options.variants.map((variant) => ({ TypeTagToUseInstead: tag(variant), Weight: 0.2 })) } } : {}),
  };
}

export const areaRow = (name: string, typeTag: string, mobs: readonly string[], key?: string) => ({
  Name: name,
  TypeTag: tag(typeTag),
  TypeTextData: nameText("DT_AreaDefinition", key ?? `SW_Area_${name}_SW_TextEntry_Name`),
  AreaDefinition: { ConditionGroups: [{ DefaultDefinition: { MobTypes: mobs.map((mob) => ({ MobType: tag(mob) })) } }] },
});

export const waveRow = (name: string, mobs: readonly string[]) => ({
  Name: name,
  SpawnWaveDefinition: { Waves: [{ Waves: mobs.map((mob) => ({ MobType: tag(mob) })) }] },
});

const EMPTY_MOBS = { definitions: [], type1: [], type2: [], type3: [], soulStated: [], variants: [], areas: [], waves: [] };
export const mobTables = (partial: Partial<Record<keyof MobTables, readonly unknown[]>>): MobTables =>
  MobTablesSchema.parse({ ...EMPTY_MOBS, ...partial });

/** A pool entry: a tag (Set, weight 1) or [tag, weight, operation]. */
export type PoolTag = string | readonly [string, number, "Set" | "Multiply"];
export function slot(name: string, guaranteed: number, rolls: number, chance: number, inherit: boolean, pool: readonly PoolTag[] = []) {
  return [
    `(TagName="SW.LootSlot.${name}")`,
    {
      NumberOfGuaranteedDrops: guaranteed,
      NumberOfRandomRolls: rolls,
      DropChancePerRoll: chance,
      InheritParentDropTable: inherit,
      DropTable: pool.length === 0 ? {} : { Tags: pool.map((entry) => {
        const [name2, Weight, Operation] = typeof entry === "string" ? [entry, 1, "Set"] : entry;
        return { TypeTag: tag(name2), Weight, Operation };
      }) },
    },
  ] as const;
}

export const lootRow = (name: string, typeTag: string, slots: readonly ReturnType<typeof slot>[], traits: readonly string[] = []) => ({
  Name: name,
  TypeTag: tag(typeTag),
  ...(traits.length > 0 ? { Traits: tags(...traits) } : {}),
  LootDropDefinition: { LootSlots: Object.fromEntries(slots) },
});

export const itemRow = (table: string, name: string, typeTag: string, traits: readonly string[] = [], idKey?: string) => ({
  Name: name,
  TypeTag: tag(typeTag),
  Traits: tags(...traits),
  TypeTextData: nameText(table, idKey ?? `SW_Item_${name}_SW_TextEntry_Name`),
});

const EMPTY_LOOT = { generic: [], mob: [], area: [], lootActor: [], quest: [], melee: [], ranged: [], armor: [], artifact: [], enchantmentBook: [], talisman: [] };
export const lootTables = (partial: Partial<Record<keyof LootTables, readonly unknown[]>>): LootTables =>
  LootTablesSchema.parse({ ...EMPTY_LOOT, ...partial });

  for (const [name, content] of Object.entries(files)) await writeFile(path.join(dir, name), content, "utf8");

- Produces（`loot-sources.ts`）：
  ```ts
  export interface SourcesInput { tables: LootTables; index: ItemIndex; namer: AreaNamer; locales: Readonly<Record<Lang, LocaleFile>>; nameKeys?: Readonly<Record<string, { namespace: string; key: string }>> }
  export function buildItemSources(input: SourcesInput): { data: ItemSourcesData; issues: Issue[] };
  ```
  規則：
  1. **區域池**：每個 `tables.area` 列（跳過 TypeTag 剛好是 `SW.Area` 的 `GlobalAreaLoot`）的 `item` slot 池 → `index.resolve` → 對每件物品加 `areas: { areaId: areaPrefix(row.TypeTag), name: namer(...) }`（以 `areaId` 去重；名稱找不到 → 英文詞＋untranslated，不略過）。
  2. **寶箱**（`tables.lootActor`）：對 `item`、`specialGear`、`talismans` slot：有自己的池 → 物品 `chests` 加 `{ id: row.Name, name, pool: "fixed" }`；沒有自己的池且 `InheritParentDropTable` 為真 → 對「出現在任何區域池的物品」加 `pool: "area"`。同一寶箱對同一物品只記一次，`fixed` 優先。
  3. **任務**（`tables.quest`）：只看 `specialGear`、`item`、`enchantedBook` slot；`NumberOfGuaranteedDrops > 0`（或必掉 0 且 `NumberOfRandomRolls > 0` 且 `DropChancePerRoll >= 1`）才算必得，否則進 `loot-quest-not-guaranteed` 並略過；池標籤展開成物品，每件加 `{ id: row.Name, name, guaranteed: true, picks? }`，`picks = { count: guaranteed, of: 展開後件數 }` 只在 `count < of` 時出現。
  4. **商人**（`tables.generic` 中 TypeTag 開頭 `SW.NPC.`）：`item` slot 的池標籤若是物品 trait 類別（`index.withTrait(tag)` 非空，例如 `SW.Item.Property.MeleeWeapon`）→ 對這些物品，扣掉具有任何 `excludedTags` 的 trait 者，加 `{ id, name, byCategory: true }`。**不展開成其他資料，也不為精確物品標籤建立商人來源。**
  5. **書池**（`tables.generic` 中 TypeTag 開頭 `SW.LootSource.EnchantmentBooks`）：`rowPool(row, "item", byTag)` 展開成書，加 `{ id: row.Name, name }`。（`EnchantmentBooksExcludeQuestBooks` 本身是排除列，不當書池：它沒有 Set 標籤，自然為空。）
  6. 每個物品的 `untranslated` = 其所有 `name` 的 `untranslated` 聯集（空則省略欄位）；各清單依 `id` 排序；沒有任何來源的物品不出現在輸出。
  7. 回報：池中標籤既不是物品／套裝標籤、又不在 `SW.Item.Currency.`、`SW.Item.Consumable.`、`SW.Item.Property.`、`SW.Item.ExceptFromDefaultLootTables`、`SW.Item.TNT` 底下 → `loot-tag-unmatched`（每個標籤一條）；`tagVariants` 的每組 → `loot-tag-case`；任何 `Name` 含 `[^A-Za-z0-9_]` 的列 → `loot-row-name`；名稱退英文 → `source-name-fallback`（每個 id 一條）。

import { existsSync, readdirSync, readFileSync } from "node:fs";
import path from "node:path";
import { describe, expect, it } from "vitest";

const ROOT = path.resolve(__dirname, "../../..");
const CONSTANTS = path.join(ROOT, "source-data", "DataTables", "UFS", "OnlineConstants.json");
const DATA = path.join(ROOT, "src", "data");
const MIN_LENGTH = 20;

const stringValues = (value: unknown): string[] => {
  if (typeof value === "string") return [value];
  if (Array.isArray(value)) return value.flatMap(stringValues);
  if (value && typeof value === "object") return Object.values(value).flatMap(stringValues);
  return [];
};

describe("telemetry constants", () => {
  it.skipIf(!existsSync(CONSTANTS))("never reach the published data", () => {
    const secrets = stringValues(JSON.parse(readFileSync(CONSTANTS, "utf8").replace(/^﻿/, ""))).filter((value) => value.length >= MIN_LENGTH);
    expect(secrets.length).toBeGreaterThan(0);
    for (const file of readdirSync(DATA).filter((name) => name.endsWith(".json"))) {
      const text = readFileSync(path.join(DATA, file), "utf8");
      for (const secret of secrets) expect(text.includes(secret), `${file} contains a telemetry value`).toBe(false);
    }
  });

  it("is never named by extractor code", () => {
    const sources = readdirSync(path.join(ROOT, "scripts", "extract")).filter((name) => name.endsWith(".ts") && name !== "table-reader.ts");
    for (const name of sources) expect(readFileSync(path.join(ROOT, "scripts", "extract", name), "utf8"), name).not.toMatch(/OnlineConstants/);
  });
});
